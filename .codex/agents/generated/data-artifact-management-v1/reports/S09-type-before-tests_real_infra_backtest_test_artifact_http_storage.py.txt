def test_http_transaction_bulk_replay_versions_private_and_dto(context):
    m = context
    client, principal = http(m)
    c = {"exchange": "binance", "market_type": "spot", "symbol": "BTCUSDT"}
    body = {"items": [{"coordinate": c, "expected_version": 0}]}
    url = "/api/backtest-artifacts/build-requests"
    headers = {"Origin": "http://testserver", "Idempotency-Key": "http-command-001"}
    assert (
        client.post(url, json=body, headers={"Idempotency-Key": "http-command-001"}).status_code
        == 403
    )
    bad = client.post(url, json={"items": [dict(body["items"][0], source={})]}, headers=headers)
    assert bad.status_code == 422
    result = client.post(url, json=body, headers=headers)
    assert result.status_code == 202, result.text
    assert client.post(url, json=body, headers=headers).json() == result.json()
    assert (
        client.post(
            url, json=body, headers=dict(headers, **{"Idempotency-Key": "http-command-002"})
        ).status_code
        == 409
    )
    assert client.get("/api/backtest-artifacts/commands/http-command-001").json() == result.json()
    request = result.json()["items"][0]["request_id"]
    assert "snapshot" not in client.get("/api/backtest-artifacts/requests/" + request).text
    principal.user_id = UserId(m["viewer"])
    assert client.get("/api/backtest-artifacts/requests/" + request).status_code == 200
    assert client.post(url, json=body, headers=headers).status_code == 403
    m["org"] = m["other"]
    assert client.get("/api/backtest-artifacts/requests/" + request).status_code == 404
    assert client.get("/api/backtest-artifacts/commands/http-command-001").status_code == 404

def slots(m):
    import hashlib

    import yaml

    from trading.contexts.backtest_artifacts.application.services.management import Coordinate

    c = Coordinate(exchange="binance", market_type="spot", symbol="BTCUSDT")
    base = m["root"] / "binance" / "spot" / "BTCUSDT"
    base.mkdir(parents=True)
    hashes = {}
    for n, slot in enumerate(("slot_a", "slot_b"), 1):
        path = base / slot
        path.mkdir()
        manifest = yaml.safe_dump({"slot_generation": n}).encode()
        (path / "manifest.yaml").write_bytes(manifest)
        (path / "prices.npy").write_bytes(b"owned disposable payload")
        hashes[slot] = hashlib.sha256(manifest).hexdigest()
        m["db"].execute(
            query="""INSERT INTO
                backtest_artifact_slot_ownership(exchange,market_type,symbol,slot,generation,manifest_sha256)
            VALUES ('binance','spot','BTCUSDT',%(slot)s,%(n)s,%(h)s)""",
            parameters={"slot": slot, "n": n, "h": hashes[slot]},
        )
    (base / "current.yaml").write_text(
        yaml.safe_dump(
            {
                "schema_version": 1,
                "asof_date": "2026-10-08",
                "published_at_utc": "2026-10-08T00:00:00Z",
                "active_slot": "slot_b",
                "slot_generation": 2,
                "manifest_sha256": hashes["slot_b"],
            }
        )
    )
    return c, base, hashes


def test_cleanup_actual_effect_current_and_unknown_slot_protected(context):

    s = m["storage"]

def test_progress_does_not_expose_foreign_derivative_totals():

    raw = {
        "series": {
            "a": {
                "phase": "base",
                "completed_units": 7,
                "total_units": 10,
                "unit": "bars",
                "progress_epoch": 1,
            },
            "b": {
                "phase": "derivatives",
                "completed_units": 999,

            },
        }

def test_pre_s06_command_body_hash_remains_replayable(context):
    from tests.real_infra.backtest.test_artifact_management import item
    from trading.contexts.backtest_artifacts.application.services.management import digest

    m = context
    spec = item(m)
    previous_hash = digest(
        {"kind": "update_now", "items": [spec.model_dump(mode="json", exclude={"admit_request"})]}
    )
    args = dict(organization_id=m["org"], actor_id=m["actor"], items=(spec,), key="pre-s06-command")
    original = m["repo"].submit(**args, command_body_digest=previous_hash)
    assert m["repo"].submit(**args) == original


def test_history_continues_past_revoked_cursor_coordinate(context):
    m = context
    client, _ = http(m)
    for index, symbol in enumerate(("BTCUSDT", "ETHUSDT")):
        response = client.post(
            "/api/backtest-artifacts/build-requests",
            json={
                "items": [
                    {
                        "coordinate": {
                            "exchange": "binance",
                            "market_type": "spot",
                            "symbol": symbol,
                        },
                        "expected_version": 0,
                    }
                ]
            },
            headers={"Origin": "http://testserver", "Idempotency-Key": f"cursor-test-{index}"},
        )
        assert response.status_code == 202, response.text
    original = m["repo"].access

    class RevokedEth:
        def require(self, **kwargs):
            from trading.contexts.backtest_artifacts.application.services.management import (
                ManagementError,
            )

            if kwargs["coordinate"].symbol == "ETHUSDT":
                raise ManagementError("coordinate_forbidden")
            return original.require(**kwargs)

    m["repo"].access = RevokedEth()
    first = client.get("/api/backtest-artifacts/requests?limit=1")
    assert first.status_code == 200 and first.json()["items"] == []
    second = client.get(
        "/api/backtest-artifacts/requests",
        params={"limit": 1, "cursor": first.json()["next_cursor"]},
    )
    assert second.status_code == 200, second.text
    assert second.json()["items"][0]["coordinate"]["symbol"] == "BTCUSDT"
