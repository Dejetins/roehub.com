def test_fair_lanes_poison_request_and_quota_independence(dispatch, tmp_path):
    from uuid import uuid4

    m = dispatch
    r = m["repo"]
    p = item(m)
    a = submit(m, (p,))["items"][0]["request_id"]
    b = submit(m, (p,), key="other-roundrobin", organization_id=m["other"])["items"][0][
        "request_id"
    ]
    # An invalid frozen catalog never poisons a mandatory base request.
    r._exec(
        "UPDATE backtest_artifact_requests SET "
        "snapshot=jsonb_set(snapshot,'{profile,mode}','\"precompute\"') WHERE request_id=%(r)s",
        r=UUID(a),
    )
    r._exec(
        "UPDATE backtest_artifact_requests SET "
        "snapshot=jsonb_set(snapshot,'{profile,catalog_indicators}','[]') WHERE request_id=%(r)s",
        r=UUID(a),
    )
    selected = r.next_request(m["installation"])
    build = r.claim_group(
        installation=m["installation"],
        request=selected,
        owner=uuid4(),
        incarnation=uuid4(),
        reservation=100,
    )
    assert build is not None
    assert r.rows(
        "SELECT state,reason FROM backtest_artifact_requests WHERE request_id=%(r)s", r=UUID(a)
    )[0] == {"state": "error", "reason": "catalog_snapshot_required"}
    assert (
        r.rows("SELECT state FROM backtest_artifact_requests WHERE request_id=%(r)s", r=UUID(b))[0][
            "state"
        ]
        == "running"
    )
    r.finish(build=build, outcome="error", reason="test_finished")
    # Failed requests do not leave a permanently selected poisoned FIFO head.
    assert r.next_request(m["installation"]) is None

def test_progress_series_cannot_regress_after_another_phase(dispatch):
    from datetime import UTC, datetime
    from uuid import uuid4

    from trading.contexts.backtest_artifacts.application.services.management import (
        ManagementError,
        Progress,
    )

    m = dispatch
    r = m["repo"]
    submit(m, (item(m),))
    build = r.claim_group(
        installation=m["installation"],
        request=r.next_request(m["installation"]),
        owner=uuid4(),
        incarnation=uuid4(),
        reservation=100,
    )
    p = Progress(
        phase="base",
        timeframe="15m",
        completed_units=4,
        total_units=None,
        unit="bars",
        observed_at=NOW,
    )
    r.observe(build=build, progress=p)
    r.observe(
        build=build,
        progress=p.model_copy(
            update={"phase": "derivatives", "observed_at": NOW + timedelta(seconds=1)}
        ),
    )
    with pytest.raises(ManagementError, match="progress_regression"):
        r.observe(
            build=build,
            progress=p.model_copy(
                update={"completed_units": 3, "observed_at": NOW + timedelta(seconds=2)}
            ),
        )
    r.finish(build=build, outcome="retry_wait", reason="transient_io")
    fresh = r.retry(
        build=build,
        owner_token=uuid4(),
        parent_incarnation=uuid4(),
        now=datetime.now(UTC) + timedelta(days=1),
    )
    assert fresh["epoch"] == 2 and fresh["progress"] == {}
    r.observe(build=fresh, progress=p.model_copy(update={"completed_units": 0}))
    assert (
        r.rows("SELECT progress FROM backtest_artifact_builds")[0]["progress"]["progress_epoch"]
        == 2
    )


def test_handoff_requires_owner_and_legacy_drain(dispatch, tmp_path):
    from apps.scheduler.backtest_artifact_publisher.wiring.modules.backtest_artifact_publisher import (  # noqa: E501
        FileBacktestArtifactPublisherHostLock,
    )
    from trading.contexts.backtest_artifacts.application.services.management import ManagementError

    m = dispatch
    r = m["repo"]
    lock = FileBacktestArtifactPublisherHostLock(tmp_path / "legacy.lock")
    r._exec(
        "UPDATE backtest_artifact_installation_settings SET capability_enabled=FALSE,"
        "paused=TRUE,legacy_handoff='pending'"
    )
    args = dict(
        organization_id=m["org"], actor_id=m["actor"], expected_version=1, host_lock=lock, now=NOW
    )
    with pytest.raises(ManagementError, match="legacy_scheduler_enabled"):
        r.record_handoff(**args, legacy_disabled=False)
    with pytest.raises(ManagementError, match="installation_forbidden"):
        r.record_handoff(**dict(args, actor_id=m["admin"]), legacy_disabled=True)
    r.record_handoff(**args, legacy_disabled=True)
    state = r.rows("SELECT * FROM backtest_artifact_installation_settings")[0]
    assert state["legacy_handoff"] == "complete" and state["capability_enabled"]
    r._exec("UPDATE backtest_artifact_installation_settings SET capability_enabled=FALSE")
    # Rollback of submission does not implicitly hand ownership back to the legacy writer.
    assert (
        r.rows("SELECT legacy_handoff FROM backtest_artifact_installation_settings")[0][
            "legacy_handoff"
        ]
        == "complete"
    )

def test_concurrent_claims_and_fifo_round_robin(dispatch):
    from concurrent.futures import ThreadPoolExecutor
    from uuid import uuid4

    m = dispatch
    r = m["repo"]
    p = item(m)
    submit(m, (p,))
    submit(m, (item(m, "ETHUSDT"),), key="fifo-second")
    submit(m, (p,), key="tenant-other", organization_id=m["other"])
    selected = r.next_request(m["installation"])

    def claim_one():
        return r.claim_group(
            installation=m["installation"],
            request=selected,
            owner=uuid4(),
            incarnation=uuid4(),
            reservation=100,
        )

    with ThreadPoolExecutor(max_workers=2) as pool:
        results = list(pool.map(lambda _: claim_one(), range(2)))
    assert sum(x is not None for x in results) == 1
    build = next(x for x in results if x)
    assert len(r.rows("SELECT * FROM backtest_artifact_builds")) == 1
    assert {
        x["organization_id"]
        for x in r.rows(
            "SELECT organization_id FROM backtest_artifact_requests WHERE state='running'"
        )
    } == {m["org"], m["other"]}
    r.finish(build=build, outcome="error", reason="test_finished")
    assert r.next_request(m["installation"])["symbol"] == "ETHUSDT"

def _claim_and_hold(dsn, root, installation, ready):
    from time import sleep
    from uuid import uuid4

    from apps.scheduler.backtest_artifact_publisher.wiring.managed_recovery import lifetime
    from tests.real_infra.backtest.test_artifact_management import Access
    from trading.contexts.backtest_artifacts.adapters.outbound.persistence.postgres.gateway import (

    )

    repo = PostgresArtifactDispatchRepository(
        gateway=PsycopgBacktestPostgresGateway(dsn=dsn), catalog=None, access=Access()
    )
    owner = uuid4()
    with lifetime(Path(root), {"owner_token": owner}):
        build = repo.claim_group(
            installation=installation,
            request=repo.next_request(installation),
            owner=owner,
            incarnation=uuid4(),
            reservation=100,
        )
        ready.put(str(build["build_id"]))

def test_actual_process_death_retains_unknown_pointer_quarantine(dispatch, tmp_path):
    import multiprocessing

    m = dispatch
    r = m["repo"]
    submit(m, (item(m),))
    _, w = worker(m, tmp_path)
    ctx = multiprocessing.get_context("spawn")
    ready = ctx.Queue()
    child = ctx.Process(
        target=_claim_and_hold, args=(m["dsn"], str(w.storage.root), m["installation"], ready)
    )
    child.start()
    try:
        build_id = ready.get(timeout=30)
        assert w.reconcile(m["installation"]) == 0
        assert r.rows("SELECT state FROM backtest_artifact_builds")[0]["state"] == "running"
        child.terminate()
        child.join(timeout=10)
        assert not child.is_alive()
        assert w.reconcile(m["installation"]) == 0
        assert r.rows("SELECT state FROM backtest_artifact_builds")[0]["state"] == "quarantined"
        directory = (
            w.storage.root
            / ".managed-attempts"
            / str(r.rows("SELECT owner_token FROM backtest_artifact_builds")[0]["owner_token"])
        )
        (directory / "receipt.json").write_text("{broken")
        assert w.reconcile(m["installation"]) == 0
        assert (
            str(r.rows("SELECT build_id FROM backtest_artifact_builds")[0]["build_id"]) == build_id
        )
    finally:
        if child.is_alive():
            child.terminate()
            child.join(timeout=10)
        ready.close()

def test_composed_retry_uses_new_epoch_and_finishes_real_work(dispatch, tmp_path):
    import errno
    from datetime import UTC, datetime

    from apps.scheduler.backtest_artifact_publisher.wiring.managed_app import (
        ManagedArtifactPublisherApp,
    )
    from apps.scheduler.backtest_artifact_publisher.wiring.modules.backtest_artifact_publisher import (  # noqa: E501
        FileBacktestArtifactPublisherHostLock,
    )

    m = dispatch
    r = m["repo"]
    p = item(m)
    p = p.model_copy(
        update={
            "source": p.source.model_copy(
                update={
                    "history": History(
                        kind="range", start_utc=START, end_utc=START + timedelta(minutes=180)
                    ),
                    "watermark": START + timedelta(minutes=180),
                }
            )
        }
    )
    submit(m, (p,))
    _, w = worker(m, tmp_path)

    class Once:
        def __init__(self):
            self.failed = False

        def read_1m_arrays(self, **kwargs):
            if not self.failed:
                self.failed = True
                raise OSError(errno.EAGAIN, "fixture transient source")
            return Source(180).read_1m_arrays(**kwargs)

    w.publisher = replace(
        w.publisher, runner=replace(w.publisher.runner, canonical_candle_reader=Once())
    )

    class Reconciler:
        def reconcile_page(self, **kwargs):
            return None

    clock = [datetime.now(UTC)]
    app = ManagedArtifactPublisherApp(
        m["installation"],
        w,
        Reconciler(),
        FileBacktestArtifactPublisherHostLock(tmp_path / "host.lock"),
        lambda: clock[0],
    )
    with pytest.raises(OSError):
        app.cycle()
    build = r.rows("SELECT * FROM backtest_artifact_builds")[0]
    assert build["state"] == "retry_wait" and build["epoch"] == 1
    assert app.cycle() == "observed"
    assert r.rows("SELECT epoch FROM backtest_artifact_builds")[0]["epoch"] == 1
    clock[0] += timedelta(minutes=6)
    assert app.cycle() == "observed"
    build = r.rows("SELECT * FROM backtest_artifact_builds")[0]
    assert build["state"] == "ready" and build["epoch"] == 2
    assert len(r.rows("SELECT * FROM backtest_artifact_attempts")) == 2
