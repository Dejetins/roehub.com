def quantile(values,p):
 values=sorted(values); index=(len(values)-1)*p; low=int(index);high=min(low+1,len(values)-1)
 return values[low]+(values[high]-values[low])*(index-low)
def counters(text):
 out={}
 for line in text.splitlines():
  parts=line.split()
  if len(parts)==2 and '=' not in parts[1]: out[parts[0]]=int(parts[1])
  elif len(parts)>2:
   for item in parts[1:]:
    k,v=item.split('=');out['io_'+k]=out.get('io_'+k,0)+int(v)
 return out
